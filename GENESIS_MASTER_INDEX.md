   ```bash
   python tools/register_npu_node.py --node-id "NPU-001" --region "REG-CODE" --entity "ENTITY-ID"
   ```

2. **Governance Initialization**
   ```bash
   python tools/init_npu_governance.py --node-id "NPU-001" --ecg-key "$ECG_KEY"
   ```

3. **License Generation**
   ```bash
   python tools/generate_synergyze_license.py --entity "EntityName" --modules "commerce,governance,identity" --term "annual"
   ```

4. **License Assignment**
   ```bash
   python tools/assign_license_to_npu.py --license "LICENSE-ID" --node-id "NPU-001"
   ```

5. **Entity Onboarding**
   ```bash
   python tools/onboard_entity.py --name "EntityName" --region "REG-CODE"
   ```

6. **Launch Entity Services**
   ```bash
   python config/entity-name/launch.py
   ```

7. **Start Genesis Dashboard**
   ```bash
   streamlit run genesis_dashboard.py --server.port 5000
   ```
