STEP 2: ONBOARD VIA STREAMLIT UI ON REPLIT

Add a main.py file like this:

import streamlit as st
import json

st.title("Genesis WTO License Launcher")

region = st.selectbox("Select Region", ["EU", "APAC", "SAARC", "The Americas", "Middle East", "Africa"])
entity_name = st.text_input("Entity Name")
hsn_code = st.text_input("HSN Code (e.g., HSN-GEN-20-03-P)")
user_count = st.number_input("User Count", value=1000)
npu_count = st.number_input("NPU Count", value=2)
duration = st.selectbox("Duration", ["monthly", "annual", "triennial", "perpetual"])

if st.button("Generate Manifest"):
    st.success("Manifest Created!")
    st.json({
        "entity": entity_name,
        "region_block": region,
        "hsn_code": hsn_code,
        "user_count": user_count,
        "npu_count": npu_count,
        "duration": duration
    })
